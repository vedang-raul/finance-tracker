Connect to Supabase

One-time setup. Credentials saved in your browser.
1. Go to supabase.com → your project
2. SQL Editor → run the query below (if not done yet)
3. Settings → API → copy URL & anon key 
-- Run once in Supabase SQL Editor

create table if not exists users (
  id uuid primary key default gen_random_uuid(),
  username text unique not null,
  password_hash text not null,
  created_at timestamptz default now()
);

create table if not exists tags (
  id uuid primary key default gen_random_uuid(),
  user_id uuid references users(id) on delete cascade,
  name text not null,
  color text not null,
  created_at timestamptz default now()
);

create table if not exists transactions (
  id uuid primary key default gen_random_uuid(),
  user_id uuid references users(id) on delete cascade,
  amount numeric not null,
  purpose text not null,
  tag_id uuid references tags(id) on delete set null,
  created_at timestamptz default now()
);

-- Allow anon access (we handle auth ourselves)
alter table users enable row level security;
alter table tags enable row level security;
alter table transactions enable row level security;

create policy "open" on users for all using (true) with check (true);
create policy "open" on tags for all using (true) with check (true);
create policy "open" on transactions for all using (true) with check (true);
